```ts
/** For demo recordings: a clock that reads `start` now and runs on in real time. Unset: the real clock. */
export function offsetClock(start: string | undefined, real: () => number = Date.now): () => Date {
  if (!start) return () => new Date(real());
  const at = Date.parse(start);
  if (!Number.isFinite(at)) throw new Error(`MIA_CLOCK_START is not a date: "${start}"`);
  const offset = at - real();
  return () => new Date(real() + offset);
}
```

```ts
export interface Mailer { send(m: { subject: string; text: string }): Promise<void> }

const RANK = { low: 0, medium: 1, high: 2 } as const;
export const MAIL_DEDUPE_MS = 6 * 3_600_000;

/** Stores every alert, and mails medium and high ones to the carer. A cry for help that cannot be mailed throws. */
export class EmailAlertSink implements AlertSink {
  private readonly sent = new Map<string, number>();
  constructor(private readonly inner: AlertSink, private readonly mailer: Mailer, private readonly o: { now: () => Date }) {}
  async raise(a: AlertInput): Promise<void> {
    await this.inner.raise(a);
    if (RANK[a.severity] < RANK.medium) return;
    const key = `${a.profileId}|${a.rule}|${a.message}`;
    const t = this.o.now().getTime();
    const last = this.sent.get(key);
    if (a.severity !== "high" && last !== undefined && t - last < MAIL_DEDUPE_MS) return;
    try {
      await this.mailer.send({
        subject: `[Mia] ${a.severity === "high" ? "URGENT: " : ""}${a.message}`,
        text: `${a.message}\n\nRule: ${a.rule}\nSeverity: ${a.severity}\nAt: ${a.at}\n\nSent by Mia (local mode, not production).`,
      });
      this.sent.set(key, t);
    } catch (err) {
      console.error(`[mia] alert mail failed (${a.severity}):`, err instanceof Error ? err.message : err);
      if (a.severity === "high") throw err;
    }
  }
}
```

```ts
import { SESv2Client, SendEmailCommand } from "@aws-sdk/client-sesv2";
import type { Mailer } from "./alerts";

/** SES v2 with the default credential chain (the EC2 instance role on AWS). Off unless both addresses are set. */
export function sesMailerFromEnv(env: Record<string, string | undefined>): Mailer | null {
  const to = env.MIA_ALERT_EMAIL_TO?.trim();
  const from = env.MIA_ALERT_EMAIL_FROM?.trim();
  if (!to || !from) return null;
  const client = new SESv2Client({ region: env.AWS_REGION ?? "eu-central-1" });
  return {
    async send({ subject, text }) {
      await client.send(new SendEmailCommand({
        FromEmailAddress: from,
        Destination: { ToAddresses: [to] },
        Content: { Simple: { Subject: { Data: subject, Charset: "UTF-8" }, Body: { Text: { Data: text, Charset: "UTF-8" } } } },
      }));
    },
  };
}
```

```bash
# One-time: SES identities, IAM role, security group, key pair, EC2 t4g.small (encrypted gp3), Elastic IP. eu-central-1.
set -euo pipefail
: "${AWS_PROFILE:?set AWS_PROFILE (e.g. mia)}"; : "${ALERT_TO:?carer email}"; : "${ALERT_FROM:?sender email}"
export AWS_REGION=eu-central-1
NAME=mia
for addr in "$ALERT_TO" "$ALERT_FROM"; do
  aws sesv2 create-email-identity --email-identity "$addr" >/dev/null 2>&1 && echo "SES: verification mail sent to $addr" || echo "SES: $addr exists"
done
aws iam create-role --role-name $NAME-ec2 --assume-role-policy-document \
  '{"Version":"2012-10-17","Statement":[{"Effect":"Allow","Principal":{"Service":"ec2.amazonaws.com"},"Action":"sts:AssumeRole"}]}' >/dev/null 2>&1 || true
aws iam put-role-policy --role-name $NAME-ec2 --policy-name ses-send --policy-document \
  "{\"Version\":\"2012-10-17\",\"Statement\":[{\"Effect\":\"Allow\",\"Action\":\"ses:SendEmail\",\"Resource\":\"*\",\"Condition\":{\"StringEquals\":{\"ses:FromAddress\":\"$ALERT_FROM\"}}}]}"
aws iam create-instance-profile --instance-profile-name $NAME-ec2 >/dev/null 2>&1 || true
aws iam add-role-to-instance-profile --instance-profile-name $NAME-ec2 --role-name $NAME-ec2 2>/dev/null || true
VPC=$(aws ec2 describe-vpcs --filters Name=isDefault,Values=true --query 'Vpcs[0].VpcId' --output text)
SG=$(aws ec2 describe-security-groups --filters Name=group-name,Values=$NAME Name=vpc-id,Values=$VPC --query 'SecurityGroups[0].GroupId' --output text)
if [ "$SG" = "None" ]; then
  SG=$(aws ec2 create-security-group --group-name $NAME --description "Mia: https + ssh from owner" --vpc-id "$VPC" --query GroupId --output text)
  aws ec2 authorize-security-group-ingress --group-id "$SG" --protocol tcp --port 443 --cidr 0.0.0.0/0
  aws ec2 authorize-security-group-ingress --group-id "$SG" --protocol tcp --port 80 --cidr 0.0.0.0/0
  aws ec2 authorize-security-group-ingress --group-id "$SG" --protocol tcp --port 22 --cidr "$(curl -s https://checkip.amazonaws.com)/32"
fi
KEY=~/.ssh/$NAME-eu-central-1.pem
[ -f "$KEY" ] || { aws ec2 create-key-pair --key-name $NAME --query KeyMaterial --output text > "$KEY"; chmod 600 "$KEY"; }
AMI=$(aws ssm get-parameter --name /aws/service/ami-amazon-linux-latest/al2023-ami-kernel-default-arm64 --query Parameter.Value --output text)
sleep 10  # instance profile propagation
ID=$(aws ec2 run-instances --image-id "$AMI" --instance-type t4g.small --key-name $NAME --security-group-ids "$SG" \

aws ec2 wait instance-running --instance-ids "$ID"
ALLOC=$(aws ec2 allocate-address --query AllocationId --output text)
aws ec2 associate-address --instance-id "$ID" --allocation-id "$ALLOC" >/dev/null
IP=$(aws ec2 describe-addresses --allocation-ids "$ALLOC" --query 'Addresses[0].PublicIp' --output text)
echo "instance $ID  ip $IP  →  https://${IP//./-}.sslip.io   ssh -i $KEY ec2-user@$IP"
```

```bash
# Sync the repo to the server and restart Mia. First run installs Node 24, pnpm, Caddy and writes /etc/mia/*.env.
set -euo pipefail

set -euo pipefail
if ! command -v caddy >/dev/null; then

sudo rsync -a --delete --exclude .mia-data /tmp/mia-src/ /opt/mia/

printf '[Service]\nEnvironmentFile=/etc/mia/caddy.env\n' | sudo tee /etc/systemd/system/caddy.service.d/mia.conf >/dev/null
sudo systemctl daemon-reload && sudo systemctl enable --now mia caddy && sudo systemctl restart mia caddy
REMOTE
```
